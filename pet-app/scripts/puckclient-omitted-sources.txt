# Files in folders PuckClient draws from that it deliberately does not compile.
#
# PuckClient names its sources one file at a time (see project.yml), so a new
# file in one of those folders is silently absent from it. check-client-sources.py
# fails on any such file that is neither listed there nor recorded here, so
# every addition is a decision somebody made rather than one nobody noticed.
#
# Adding a line here means: the pet app owns this, the chat window has no part
# in it. If that stops being true, move it into project.yml instead.

# The pet's own process: its delegate, menu bar, and the toys it carries.
# PuckClient has no overlay window and no Dock-less menu bar item.
Puck/App/MenuBarController.swift
Puck/App/PuckApp.swift
Puck/App/ToyPresentation.swift
Puck/App/ToyThumbnail.swift

# The server half of the bridge. Puck hosts the socket and decides who each
# message is for; PuckClient only dials in, so it compiles the client half.
Puck/Bridge/BridgeMessageRouter.swift
Puck/Bridge/BridgeServer.swift
Puck/Bridge/ClientRelay.swift
Puck/Bridge/EventRouter.swift

# Screen-recording and accessibility onboarding. Only the pet needs those

Puck/Diagnostics/PermissionOnboarding.swift
Puck/Diagnostics/PermissionPromptPolicy.swift
